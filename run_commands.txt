# Run commands for the Doctor Appointment Booking System (Blockchain Edition)

# 1. Install dependencies (pnpm is required)
pnpm install

# 2. Create .env.local with required variables (see SETUP.md)
# Example: copy .env.example or create manually
# cp .env.example .env.local

# 3. Start development server
pnpm dev

# 4. Build for production
pnpm build

# 5. Run production server
pnpm start

# 6. Lint
pnpm lint
